where-are-we · apt repository

One line, with upgrades:

curl -fsSL https://ngavrish.github.io/where-are-we/install.sh | sh

Or a single package, no repository:

curl -fsSLO https://github.com/ngavrish/where-are-we/releases/latest/download/where-are-we_all.deb
sudo dpkg -i where-are-we_all.deb

Or by hand:

curl -fsSL https://ngavrish.github.io/where-are-we/apt-key.asc \
  | sudo gpg --dearmor -o /usr/share/keyrings/where-are-we.gpg

echo "deb [signed-by=/usr/share/keyrings/where-are-we.gpg] \
https://ngavrish.github.io/where-are-we stable main" \
  | sudo tee /etc/apt/sources.list.d/where-are-we.list

sudo apt-get update && sudo apt-get install where-are-we
